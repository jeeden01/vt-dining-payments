<script>
// Toggle expandable cards for both desktop mouse clicks and mobile taps
document.querySelectorAll('.card').forEach(card => {
card.addEventListener('click', (e) => {
// Prevent toggle if user is trying to highlight/select text
if (window.getSelection().toString().length > 0) return;
card.classList.toggle('open');
});
});
// Category Filter Buttons
function filterCategory(cat, btn) {
document.querySelectorAll('.filter-btn').forEach(b => b.classList.remove('active'));
btn.classList.add('active');
document.querySelectorAll('.card').forEach(card => {
if (cat === 'all' || card.getAttribute('data-category') === cat) {
card.style.display = 'block';
} else {
card.style.display = 'none';
}
});
}
// Search Bar Filter
function filterDining() {
let input = document.getElementById('searchInput').value.toLowerCase();
document.querySelectorAll('.card').forEach(card => {
let keywords = card.getAttribute('data-keywords');
if (keywords.toLowerCase().includes(input)) {
card.style.display = 'block';
} else {
card.style.display = 'none';
}
});
}
